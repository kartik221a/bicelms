// Compute Quarter-over-Quarter Variance & Cohort Retention
WITH quarterly_cohorts AS (
SELECT cohort_id, date_trunc('quarter', transaction_date) AS qtr,
SUM(net_revenue) OVER (PARTITION BY cohort_id ORDER BY transaction_date) AS cumulative_arr
FROM production.corporate_ledger
)
SELECT * FROM quarterly_cohorts WHERE cumulative_arr > 250000;
